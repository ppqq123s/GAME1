        let currentDishIndex = 0;
        let addedCount = 0;

        // 初始化游戏
        function initGame() {
            const dish = dishes[currentDishIndex];
            document.getElementById('dishName').textContent = dish.name;
            document.getElementById('potEmoji').textContent = '🔥';
            document.getElementById('progressBar').style.width = '0%';
            document.getElementById('progressText').textContent = '0%';
            addedCount = 0;
            
            // 重置食材状态
            dish.ingredients.forEach(i => i.added = false);
            
            // 渲染食材
            renderIngredients();
        }

        // 渲染食材
        function renderIngredients() {
            const container = document.getElementById('ingredients');
            const dish = dishes[currentDishIndex];
            
            container.innerHTML = '';
            dish.ingredients.forEach((ing, idx) => {
                const div = document.createElement('div');
                div.className = `ingredient flex flex-col items-center p-3 rounded-2xl ${ing.added ? 'bg-green-100 opacity-50' : 'bg-white shadow-lg'}`;
                div.innerHTML = `
                    <span class="text-3xl mb-1">${ing.emoji}</span>
                    <span class="text-xs text-amber-700 font-medium">${ing.name}</span>
                `;
                
                if (!ing.added) {
                    div.onclick = (e) => addIngredient(idx, e);
                }
                
                container.appendChild(div);
            });
        }

        // 添加食材
        function addIngredient(idx, e) {
            const dish = dishes[currentDishIndex];
            if (dish.ingredients[idx].added) return;
            
            dish.ingredients[idx].added = true;
            addedCount++;
            
            // 点击特效
            createClickEffect(e);
            
            // 更新进度
            const progress = Math.round((addedCount / dish.ingredients.length) * 100);
            document.getElementById('progressBar').style.width = progress + '%';
            document.getElementById('progressText').textContent = progress + '%';
            
            // 更新锅里的内容
            const addedEmojis = dish.ingredients.filter(i => i.added).map(i => i.emoji);
            document.getElementById('potEmoji').textContent = addedEmojis.slice(-3).join(' ');
            
            // 重新渲染
            renderIngredients();
            
            // 检查是否完成
            if (addedCount === dish.ingredients.length) {
                setTimeout(showComplete, 500);
            }
        }

        // 点击特效
        function createClickEffect(e) {
            const effect = document.createElement('div');
            effect.className = 'click-effect fixed w-16 h-16 rounded-full bg-green-300/50';
            effect.style.left = (e.clientX - 32) + 'px';
            effect.style.top = (e.clientY - 32) + 'px';
            document.body.appendChild(effect);
            setTimeout(() => effect.remove(), 500);
        }

        // 显示完成
        function showComplete() {
            const dish = dishes[currentDishIndex];
            document.getElementById('dishEmoji').textContent = dish.emoji;
            document.getElementById('completeDishName').textContent = dish.name;
            document.getElementById('completeModal').classList.remove('hidden');
        }

        // 下一道菜
        function nextDish() {
            currentDishIndex = (currentDishIndex + 1) % dishes.length;
            document.getElementById('completeModal').classList.add('hidden');
            initGame();
        }

        // 重新开始
        function restartGame() {
            currentDishIndex = 0;
            document.getElementById('completeModal').classList.add('hidden');
            initGame();
        }

        // 启动游戏
        initGame();
